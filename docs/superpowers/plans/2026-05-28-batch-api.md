# Anthropic Batch API for Discovery Runs

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Route Stage 2 relevance checks in discovery runs through Anthropic's Message Batches API for a flat 50% cost reduction, while keeping manual analyses on the synchronous API for real-time response.

**Architecture:** A new `run_discovery_batch()` entry point fetches HN/Reed/Adzuna jobs, runs Stage 1 (free, synchronous), bundles all Stage-1-passing jobs into a single Anthropic batch request, polls until the batch completes (within 24h), then processes results synchronously (Stage 2 interpretation → Phase 1 scoring for relevant jobs). A new `DiscoveryBatch` model tracks batch lifecycle. Manual analyses (`POST /api/analyse`) remain unchanged — they use the synchronous `tracked_call()` path. A new route `POST /api/discovery/run/batch` triggers batch mode; the existing `POST /api/discovery/run` stays unchanged for synchronous mode.

**Tech Stack:** `anthropic.AsyncAnthropic.beta.messages.batches` (Anthropic Batch SDK), SQLAlchemy 2.0 async, FastAPI background tasks, SQLite

```python
from __future__ import annotations

COST_PER_MILLION: dict[str, dict[str, float]] = {
    "claude-haiku-4-5-20251001": {
        "input": 0.80,
        "output": 4.00,
        "cache_write": 1.00,   # 1.25× input
        "cache_read": 0.08,    # 0.10× input
    },
    "claude-sonnet-4-6": {
        "input": 3.00,
        "output": 15.00,
        "cache_write": 3.75,   # 1.25× input
        "cache_read": 0.30,    # 0.10× input
    },
}

_FALLBACK = {"input": 3.00, "output": 15.00, "cache_write": 3.75, "cache_read": 0.30}

_BATCH_DISCOUNT = 0.5  # Anthropic Message Batches: 50% off all token types


def calculate_cost(
    model: str,
    input_tokens: int,
    output_tokens: int,
    cache_creation_tokens: int = 0,
    cache_read_tokens: int = 0,
    batch: bool = False,
) -> float:
    """Compute total cost in USD.

    batch=True applies the Anthropic Message Batches 50% discount to all token types.
    """
    rates = COST_PER_MILLION.get(model, _FALLBACK)
    discount = _BATCH_DISCOUNT if batch else 1.0
    return discount * (
        input_tokens * rates["input"]
        + output_tokens * rates["output"]
        + cache_creation_tokens * rates["cache_write"]
        + cache_read_tokens * rates["cache_read"]
    ) / 1_000_000
```

```python
"""Anthropic Message Batches integration for Stage 2 discovery checks.

Provides three functions:
- submit_stage2_batch(): submit a batch, return Anthropic batch ID
- poll_batch(): check processing status
- process_batch_results(): stream results, write LLMCall rows, return Stage2Result map
"""
from __future__ import annotations

import json
import logging
from datetime import datetime, timezone
from typing import Any

import anthropic
from sqlalchemy.ext.asyncio import AsyncSession

from backend.agents.base import HAIKU
from backend.models import DiscoveryBatch, LLMCall
from backend.services.cost_calculator import calculate_cost
from backend.services.discovery import Stage2Result

logger = logging.getLogger(__name__)


async def submit_stage2_batch(
    client: anthropic.AsyncAnthropic,
    run_id: str,
    jobs: list[tuple[str, str]],   # [(job_id, raw_text), ...]
    system_prompt: str,
    *,
    db: AsyncSession,
) -> str:
    """Submit a Stage 2 batch to Anthropic. Returns the Anthropic batch ID.

    Args:
        client: Anthropic async client
        run_id: DiscoveryRun.id — used to link the DiscoveryBatch row
        jobs: list of (job_id, raw_text) — job_id becomes the custom_id prefix
        system_prompt: the static system prompt (candidate profile + instructions)
        db: AsyncSession for writing DiscoveryBatch row

    Returns:
        Anthropic batch ID (e.g. "msgbatch_01abc...")
    """
    requests = [
        {
            "custom_id": f"job-{job_id}",
            "params": {
                "model": HAIKU,
                "max_tokens": 200,
                "system": [
                    {
                        "type": "text",
                        "text": system_prompt,
                        "cache_control": {"type": "ephemeral"},
                    }
                ],
                "messages": [
                    {"role": "user", "content": f"Job posting:\n{raw_text[:3000]}"}
                ],
            },
        }
        for job_id, raw_text in jobs
    ]

    batch = await client.beta.messages.batches.create(requests=requests)

    db_batch = DiscoveryBatch(
        run_id=run_id,
        anthropic_batch_id=batch.id,
        status="processing",
        jobs_submitted=len(jobs),
        created_at=datetime.now(timezone.utc),
    )
    db.add(db_batch)
    await db.commit()

    logger.info("Submitted Stage 2 batch %s with %d jobs for run %s", batch.id, len(jobs), run_id)
    return batch.id


async def poll_batch(client: anthropic.AsyncAnthropic, batch_id: str) -> str:
    """Check Anthropic batch status. Returns 'processing', 'ended', or 'failed'."""
    status = await client.beta.messages.batches.retrieve(batch_id)
    if status.processing_status == "ended":
        return "ended"
    if status.processing_status in ("canceling", "canceled"):
        return "failed"
    return "processing"


async def process_batch_results(
    client: anthropic.AsyncAnthropic,
    batch_id: str,
    *,
    run_id: str,
    db: AsyncSession,
) -> dict[str, Stage2Result | None]:
    """Stream batch results, write LLMCall rows at batch pricing, return Stage2Result per job_id.

    Returns:
        Dict mapping job_id → Stage2Result (or None if the item errored).
        The "job-{job_id}" prefix is stripped from custom_ids.
    """
    results: dict[str, Stage2Result | None] = {}
    succeeded = 0
    failed = 0

    async for item in client.beta.messages.batches.results(batch_id):
        # custom_id is "job-{job_id}"
        job_id = item.custom_id.removeprefix("job-")

        if item.result.type == "errored":
            logger.warning(
                "Batch item %s errored: %s — %s",
                item.custom_id,
                item.result.error.type,
                item.result.error.message,
            )
            results[job_id] = None
            failed += 1
            continue

        # Parse Stage2Result from response
        msg = item.result.message
        try:
            raw_text = msg.content[0].text.strip()
            start, end = raw_text.find("{"), raw_text.rfind("}") + 1
            data = json.loads(raw_text[start:end])
            stage2 = Stage2Result(
                relevant=bool(data.get("relevant", False)),
                reason=data.get("reason", ""),
                title=data.get("title", ""),
                company=data.get("company", ""),
                location=data.get("location"),
            )
            results[job_id] = stage2
        except Exception as exc:
            logger.warning("Failed to parse Stage2Result for %s: %s", item.custom_id, exc)
            results[job_id] = None
            failed += 1
            continue

        # Write LLMCall row with batch=True pricing
        usage = msg.usage
        cache_creation = int(getattr(usage, "cache_creation_input_tokens", None) or 0)
        cache_read = int(getattr(usage, "cache_read_input_tokens", None) or 0)
        cost = calculate_cost(
            HAIKU,
            usage.input_tokens,
            usage.output_tokens,
            cache_creation_tokens=cache_creation,
            cache_read_tokens=cache_read,
            batch=True,
        )
        try:
            db.add(
                LLMCall(
                    agent_name="stage2_haiku_batch",
                    model=HAIKU,
                    input_tokens=usage.input_tokens,
                    output_tokens=usage.output_tokens,
                    cost_usd=cost,
                    latency_ms=0,  # batch calls have no per-call latency
                    cache_hit=False,
                    cache_creation_tokens=cache_creation,
                    cache_read_tokens=cache_read,
                    run_id=run_id,
                    created_at=datetime.now(timezone.utc),
                )
            )
            await db.commit()
        except Exception as exc:
            logger.warning("Failed to write LLMCall for batch item %s: %s", item.custom_id, exc)
            try:
                await db.rollback()
            except Exception:
                pass

        succeeded += 1

    # Update DiscoveryBatch row with final counts
    from sqlalchemy import select, update
    await db.execute(
        update(DiscoveryBatch)
        .where(DiscoveryBatch.anthropic_batch_id == batch_id)
        .values(
            status="ended",
            jobs_succeeded=succeeded,
            jobs_failed=failed,
            completed_at=datetime.now(timezone.utc),
        )
    )
    await db.commit()

    return results
```

```python
async def _run_discovery_batch_task(run_id: str, source: str) -> None:
    """Background task: fetch jobs, Stage 1 filter, submit Stage 2 batch, poll, process results."""
    import asyncio

    from backend.services.batch_processor import (
        poll_batch,
        process_batch_results,
        submit_stage2_batch,
    )

    try:
        async with SessionLocal() as db:
            await db.execute(
                update(DiscoveryRun).where(DiscoveryRun.id == run_id).values(status="running")
            )
            await db.commit()

            profiles = _load_search_profiles()
            profile = await get_or_build_profile(db)
            compact = build_compact_profile(profile.yaml_data, profile.cv_text)

            # Fetch raw jobs from source
            if source == "reed":
                all_roles = [r for p in profiles for r in p.target_roles]
                keywords = " ".join(all_roles[:3]) if all_roles else "software engineer"
                all_locations = [loc for p in profiles for loc in p.allowed_locations]
                location = all_locations[0] if all_locations else ""
                raw_jobs = await fetch_reed_jobs(keywords, location)
            elif source == "adzuna":
                all_roles = [r for p in profiles for r in p.target_roles]
                keywords = " ".join(all_roles[:3]) if all_roles else "software engineer"
                all_locations = [loc for p in profiles for loc in p.allowed_locations]
                location = all_locations[0] if all_locations else ""
                raw_jobs = await fetch_adzuna_jobs(keywords, location)
            else:
                raw_jobs = await fetch_hn_jobs()

            await db.execute(
                update(DiscoveryRun)
                .where(DiscoveryRun.id == run_id)
                .values(jobs_found=len(raw_jobs))
            )
            await db.commit()

            # Stage 1: keyword filter (free, synchronous)
            # Also insert Job rows immediately so they're queryable
            stage1_jobs: list[tuple[str, str]] = []  # [(job_id, raw_text)]
            for raw in raw_jobs:
                existing = (
                    await db.execute(select(Job).where(Job.dedup_hash == raw.dedup_hash))
                ).scalar_one_or_none()
                if existing is not None:
                    continue
                job = Job(
                    sources=json.dumps([source]),
                    source_id=raw.source_id,
                    source_url=raw.source_url,
                    raw_text=raw.raw_text,
                    dedup_hash=raw.dedup_hash,
                    state="discovered",
                    discovery_run_id=run_id,
                )
                db.add(job)
                await db.flush()

                if not _stage1_pass(raw.raw_text, profiles):
                    await db.execute(
                        update(Job).where(Job.id == job.id).values(state="filtered")
                    )
                else:
                    stage1_jobs.append((job.id, raw.raw_text))

            await db.commit()
            await db.execute(
                update(DiscoveryRun)
                .where(DiscoveryRun.id == run_id)
                .values(jobs_passed_stage1=len(stage1_jobs))
            )
            await db.commit()

            if not stage1_jobs:
                await db.execute(
                    update(DiscoveryRun)
                    .where(DiscoveryRun.id == run_id)
                    .values(status="complete", completed_at=datetime.now(timezone.utc))
                )
                await db.commit()
                return

            # Stage 2: submit batch
            system_prompt = (
                "You are evaluating job postings for a candidate.\n\n"
                f"Candidate summary:\n{compact[:1000]}\n\n"
                "Evaluate if the job posting is relevant to this candidate. "
                'Respond with ONLY valid JSON: {"relevant": true/false, "reason": "one sentence", '
                '"title": "job title or empty string", "company": "company name or empty string", '
                '"location": "city/remote or null"}'
            )
            batch_id = await submit_stage2_batch(
                _anthropic_client, run_id, stage1_jobs, system_prompt, db=db
            )

    except Exception as exc:
        logger.error("Batch discovery run %s setup failed: %s", run_id, exc, exc_info=True)

            await db.execute(
                update(DiscoveryRun)
                .where(DiscoveryRun.id == run_id)
                .values(status="failed", completed_at=datetime.now(timezone.utc))
            )
            await db.commit()
        return

    # Poll until batch completes (poll every 60s, up to 25h)
    for attempt in range(1500):
        await asyncio.sleep(60)
        try:
            poll_status = await poll_batch(_anthropic_client, batch_id)
        except Exception as exc:
            logger.warning("Poll attempt %d failed for batch %s: %s", attempt, batch_id, exc)
            continue
        if poll_status == "ended":
            break
        if poll_status == "failed":
            async with SessionLocal() as db:
                await db.execute(
                    update(DiscoveryRun)
                    .where(DiscoveryRun.id == run_id)
                    .values(status="failed", completed_at=datetime.now(timezone.utc))
                )
                await db.commit()
            return
    else:
        logger.error("Batch %s did not complete within 25 hours — giving up", batch_id)
        async with SessionLocal() as db:
            await db.execute(
                update(DiscoveryRun)
                .where(DiscoveryRun.id == run_id)
                .values(status="failed", completed_at=datetime.now(timezone.utc))
            )
            await db.commit()
        return

    # Process results and run Phase 1 synchronously for relevant jobs
    async with SessionLocal() as db:
        try:
            stage2_results = await process_batch_results(
                _anthropic_client, batch_id, run_id=run_id, db=db
            )
        except Exception as exc:

            await db.execute(

                .where(DiscoveryRun.id == run_id)

                        state="filtered",
                        title=s2.title if s2 else "",
                        company=s2.company if s2 else "",
                        location=s2.location if s2 else None,
                    )
                )
                await db.commit()
                continue

            await db.execute(
                update(Job).where(Job.id == job_id).values(
                    title=s2.title, company=s2.company, location=s2.location
                )
            )
            await db.commit()
            passed_stage2 += 1

            # Phase 1 scoring (synchronous, Haiku) — small subset so cost is low
            job_row = (await db.execute(select(Job).where(Job.id == job_id))).scalar_one_or_none()
            if job_row is None:
                continue
            try:
                result = await _run_phase1(
                    job_row.raw_text, profile, db, job_id=job_id, run_id=run_id, model=HAIKU
                )
                matched = _match_profiles(result.score, profiles)
                await db.execute(
                    update(Job).where(Job.id == job_id).values(
                        relevance_score=result.score,
                        matched_profiles=json.dumps(matched),
                        state="scored",
                    )
                )
                await db.commit()
                scored += 1
            except Exception as exc:
                logger.warning("Phase 1 failed for job %s: %s", job_id, exc)
                await db.execute(update(Job).where(Job.id == job_id).values(state="filtered"))
                await db.commit()

        await db.execute(
            update(DiscoveryRun)
            .where(DiscoveryRun.id == run_id)
            .values(
                jobs_passed_stage2=passed_stage2,
                jobs_scored=scored,
                status="complete",
                completed_at=datetime.now(timezone.utc),
            )
        )
        await db.commit()


async def run_discovery_batch(source: str, db: AsyncSession) -> str:
    """Public entry point. Creates DiscoveryRun, fires batch background task, returns run_id."""
    run = DiscoveryRun(
        source=f"{source}_batch",
        triggered_by="manual",
        status="pending",
        started_at=datetime.now(timezone.utc),
    )
    db.add(run)
    await db.commit()
    asyncio.create_task(_run_discovery_batch_task(run.id, source))
    return run.id
```
